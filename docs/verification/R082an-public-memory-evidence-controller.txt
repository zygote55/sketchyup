import hashlib
import json
import shutil
import subprocess
from pathlib import Path

main = Path('/home/keith/projects/sketchyup')
work = Path('/home/keith/projects/sketchyup-public-memory-publication')
stage = Path(__file__).parent
build = main / 'build/public-rendered-history-r082an2'


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


freeze = json.loads((build / 'source-freeze.json').read_text())
qualified = json.loads((build / 'result.json').read_text())
assert qualified['passed'] and qualified['normalCases'] == 12
assert qualified['releaseParserCases'] == qualified['sanitizerParserCases'] == 1
assert qualified['restorationVerified'] and qualified['earlierPrivateParserFailureVerified']
assert freeze['source'] == '60bda06e37afe51de911dd51d016df1a789597d3'
assert qualified['sourceFreeze'] == freeze
inputs = {**freeze['compiledRuntimeSha256'], **freeze['testSourceSha256']}
assert len(freeze['compiledRuntimeSha256']) == 478 and len(freeze['testSourceSha256']) == 303
for name, wanted in inputs.items():
    assert sha(work / name) == wanted, name
artifact = main / 'build/local-checks/r082an2-public-rendered-history/capture/real_model_benchmark'
assert sha(artifact) == qualified['binarySha256']
assert sha(work / 'docs/decisions/0141-forward-memory-budget.json') == '5f4c4a11f9f1d8a0a7b11b7e417e5036baa7a64d7e0c5afb15a80368ba8fe8dd'
proof = {'qualifiedSource': freeze['source'], 'runtimeInputs': 478, 'testInputs': 303,
         'runtimeAndTestSha256': inputs, 'binarySha256': qualified['binarySha256'],
         'softwareCases': 12, 'normalParserCases': 1, 'sanitizerParserCases': 1,
         'sanitizerScope': 'The identical public DRM parser and parser test only; not the full application.',
         'parserLeakDetection': True, 'earlierPrivateParserNegativeAssertionVerified': True,
         'defaultFixtureAndRendererPreserved': True, 'publicCapacityLimitsUnchanged': True,
         'fullPerformanceGateAccepted': False, 'releaseAccepted': False, 'freshFullCiRequired': True,
         'sourceFreezeSha256': sha(build / 'source-freeze.json'), 'buildResultSha256': sha(build / 'result.json'),
         'prospectiveMemoryBudgetSha256': sha(work / 'docs/decisions/0141-forward-memory-budget.json'),
         'platforms': {}}
for platform, tag in (('Intel', 'r082at2'), ('AMD', 'r082aq')):
    root = main / ('build/public-memory-' + tag)
    result = json.loads((root / 'result.json').read_text())
    campaign = json.loads((root / 'freeze.json').read_text())
    assert result['source'] == freeze['source'] and result['binarySha256'] == proof['binarySha256']
    assert result['exactProductionRuntime'] and not result['passed'] and len(result['cases']) == 10
    assert campaign['sourceFreezeSha256'] == proof['sourceFreezeSha256']
    for row in result['cases']:
        label = row['label']
        assert row['fixtureSha256'] == campaign['goldenFixtureSha256'][label]
        assert sha(root / 'evidence' / (label + '.json')) == row['reportSha256']
        assert sha(root / 'evidence' / (label + '-limits.json')) == row['limitsSha256']
        assert row['completeDriverSamples'] and row['driverSamples'] == 67
        assert row['memoryWithinDeclaredScope']
        assert all(row['planChecks'][key] for key in ('selectionWithinPlan', 'inferenceWithinPlan', 'editFeedbackWithinPlan'))
        if platform == 'AMD':
            assert row['timingWithinPlan']
    assert json.loads((root / 'evidence/repeated-10000-1-exit.json').read_text())['exitCode'] == 1
    assert 'Component placement exceeds document editing limits' in (root / 'evidence/repeated-10000-1.stderr').read_text()
    proof['platforms'][platform] = {
        'supportedCasesCompleted': 10, 'campaignComplete': False, 'experimental1mPublicCapacityRejected': True,
        'allSupportedMemoryWithinDeclaredScope': all(x['memoryWithinDeclaredScope'] for x in result['cases']),
        'allSupportedTimingWithinPlan': all(x['timingWithinPlan'] for x in result['cases']),
        'driverObservations': sum(x['driverSamples'] for x in result['cases']), 'cases': result['cases'],
        'originalResultSha256': sha(root / 'result.json'), 'campaignFreezeSha256': sha(root / 'freeze.json')}
docs = work / 'docs/verification'
(docs / 'R082an-public-memory-result.json').write_text(json.dumps(proof, indent=2) + '\n')
shutil.copyfile(Path(__file__), docs / 'R082an-public-memory-evidence-controller.txt')
shutil.copyfile(main / 'build/local-checks/r082an2-public-rendered-history/output.log', docs / 'R082an-public-memory-build.txt')
tables = []
for platform, result in proof['platforms'].items():
    for row in result['cases']:
        if '-1000-' not in row['label']:
            continue
        tables.append(f"| {platform} | {row['label']} | {row['viewportGpuCompleteP95Ms']:.3f} | {row['editToViewportGpuCompleteP95Ms']:.3f} | {row['peakRssBytes']/1024**2:.1f} | {row['driverResidentUpperBoundMaxBytes']/1024**2:.1f} |")

(stage / 'verified-evidence.json').write_text(json.dumps(proof, indent=2) + '\n')
