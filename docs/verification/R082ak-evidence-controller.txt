def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def read(root, name):
    return json.loads((root / name).read_text())

def git(*args):
    return subprocess.check_output(['git', '-C', str(work), *args], text=True).strip()

assert git('rev-parse', 'HEAD') == source
assert not git('status', '--porcelain')
nf = read(normal, 'source-freeze.json')
nr = read(normal, 'build-result.json')
sf = read(san, 'source-freeze.json')
sr = read(san, 'result.json')

gr = read(gpu, 'result.json')

assert nr['passed'] and nr['restorationVerified'] and len(nr['cases']) == 4

assert sr['worker']['passed'] and sr['worker']['memorySwapPeakBytes'] == 0

assert gr['productionLimits'] and gr['allMeasuredOperationsUnderFiveSeconds']
assert not gr['releaseAcceptance']
assert gr['binarySha256'] == nr['binarySha256'] == gf['artifacts']['bin/file_operation_input_tests']
assert gr['fixtureSha256'] == nf['fixtureSha256'] == gf['artifacts']['fixture/canonical.sketchyup']

    assert sha(work / name) == wanted, name

assert sha(main / 'build/verification-scripts/build-public-file-progress-r082ak.py') == nf['controllerSha256']

rows = []
