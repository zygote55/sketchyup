cmake_minimum_required(VERSION 3.25)
project(SketchyUp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SKETCHYUP_BUILD_DESKTOP "Build the native desktop spike" ON)
option(SKETCHYUP_BUILD_CLI "Build the Qt Core command-line driver" ON)
option(SKETCHYUP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SKETCHYUP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(clipper2 STATIC
  third_party/clipper2/src/clipper.engine.cpp
  third_party/clipper2/src/clipper.offset.cpp
  third_party/clipper2/src/clipper.rectclip.cpp
  third_party/clipper2/src/clipper.triangulation.cpp)
target_include_directories(clipper2 PUBLIC third_party/clipper2/include)
add_library(sketchyup_core src/core/model.cpp src/core/transform.cpp src/geometry/surface.cpp src/geometry/arrangement.cpp src/geometry/topology.cpp src/geometry/planar.cpp src/geometry/cleanup.cpp src/geometry/push_pull.cpp)
target_include_directories(sketchyup_core PUBLIC src)
target_link_libraries(sketchyup_core PRIVATE clipper2)
target_compile_options(sketchyup_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE sketchyup_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(scene_tests tests/scene_tests.cpp)
  target_link_libraries(scene_tests PRIVATE sketchyup_core)
  add_test(NAME scene COMMAND scene_tests)
  add_executable(push_pull_tests tests/push_pull_tests.cpp)
  target_link_libraries(push_pull_tests PRIVATE sketchyup_core)
  add_test(NAME push_pull COMMAND push_pull_tests)
  add_executable(cleanup_tests tests/cleanup_tests.cpp)
  target_link_libraries(cleanup_tests PRIVATE sketchyup_core)
  add_test(NAME cleanup COMMAND cleanup_tests)
  add_executable(planar_tests tests/planar_tests.cpp)
  target_link_libraries(planar_tests PRIVATE sketchyup_core)
  add_test(NAME planar COMMAND planar_tests)
  add_executable(topology_tests tests/topology_tests.cpp)
  target_link_libraries(topology_tests PRIVATE sketchyup_core)
  add_test(NAME topology COMMAND topology_tests)
  add_executable(arrangement_tests tests/arrangement_tests.cpp)
  target_link_libraries(arrangement_tests PRIVATE sketchyup_core)
  add_test(NAME arrangements COMMAND arrangement_tests)
endif()
if(SKETCHYUP_BUILD_DESKTOP OR SKETCHYUP_BUILD_CLI)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
  add_library(sketchyup_io src/io/document_io.cpp src/io/container.cpp src/io/save.cpp src/automation/commands.cpp src/automation/catalog.cpp)
  target_link_libraries(sketchyup_io PUBLIC sketchyup_core Qt6::Core)
  target_compile_options(sketchyup_io PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(command_tests tests/command_tests.cpp)
    target_link_libraries(command_tests PRIVATE sketchyup_io)
    add_test(NAME commands COMMAND command_tests)
    add_executable(persistence_tests tests/persistence_tests.cpp)
    target_link_libraries(persistence_tests PRIVATE sketchyup_io)
    target_compile_definitions(persistence_tests PRIVATE SKETCHYUP_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME persistence COMMAND persistence_tests)
    add_executable(persistence_benchmark tests/persistence_benchmark.cpp)
    target_link_libraries(persistence_benchmark PRIVATE sketchyup_io)
  endif()
endif()
if(SKETCHYUP_BUILD_CLI)
  add_executable(sketchyup-cli src/automation/main.cpp)
  target_link_libraries(sketchyup-cli PRIVATE sketchyup_io)
  install(TARGETS sketchyup-cli RUNTIME DESTINATION bin)
endif()
if(SKETCHYUP_BUILD_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets OpenGLWidgets Test)
  set(CMAKE_AUTOMOC ON)
  add_library(sketchyup_desktop src/app/viewport.cpp src/app/window.cpp)
  target_link_libraries(sketchyup_desktop PUBLIC sketchyup_io Qt6::Widgets Qt6::OpenGLWidgets)
  target_compile_options(sketchyup_desktop PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(sketchyup src/app/main.cpp)
  target_link_libraries(sketchyup PRIVATE sketchyup_desktop)
  if(BUILD_TESTING)
    add_executable(desktop_tests tests/desktop_tests.cpp)
    target_link_libraries(desktop_tests PRIVATE sketchyup_desktop Qt6::Test)
    add_executable(interaction_tests tests/interaction_tests.cpp)
    target_link_libraries(interaction_tests PRIVATE sketchyup_desktop Qt6::Test)
    add_executable(shell_tests tests/shell_tests.cpp)
    target_link_libraries(shell_tests PRIVATE sketchyup_desktop Qt6::Test)
    add_executable(dialog_tests tests/dialog_tests.cpp)
    target_link_libraries(dialog_tests PRIVATE sketchyup_desktop Qt6::Test)
    add_executable(viewport_tests tests/viewport_tests.cpp tests/viewport_topology.cpp)
    target_link_libraries(viewport_tests PRIVATE sketchyup_desktop Qt6::Test)
    add_test(NAME desktop COMMAND desktop_tests)
    set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  install(TARGETS sketchyup RUNTIME DESTINATION bin)
endif()
install(FILES LICENSE DESTINATION share/licenses/sketchyup)
install(FILES third_party/clipper2/LICENSE DESTINATION share/licenses/sketchyup RENAME Clipper2-LICENSE)
if(SKETCHYUP_BUILD_DESKTOP)
  install(FILES resources/io.sketchyup.SketchyUp.desktop DESTINATION share/applications)
  install(FILES resources/io.sketchyup.SketchyUp.svg DESTINATION share/icons/hicolor/scalable/apps)
  install(FILES resources/io.sketchyup.SketchyUp.xml DESTINATION share/mime/packages)
endif()
